  var navF=function(i){ return frozen&&i>stopI?frozen(i):navAt(i); };
  var investAt=function(i){
    if((endBar!=null&&i>=endBar)||(stopI!=null&&i>stopI)) return null;
    /* 수동 정리 뒤 아직 원본이 새로 진입하지 않은 동안(현금 상태)의 입금은 다음 진입부터. 그 밖에는 판단 방식대로 */
    var inCash=mans.some(function(m){ if(m>i) return false; for(var k=0;k<entries.length;k++) if(entries[k]>m) return entries[k]>i; return true; });
    if(immediate&&!inCash) return i;
    for(var k=0;k<entries.length;k++) if(inCash?entries[k]>i:entries[k]>=i) return entries[k];
    return null;
  };
  var ev=led.map(function(e,idx){ var i=e.i!=null?e.i:(idx===0?startI:T); return {type:e.type,amt:e.amt,i:i,inv:e.type==='add'?investAt(i):null,idx:idx}; });
  /* 봉 b까지 원장을 재생한 내 계좌: 좌수(u)와 그 매입 원금(cb), 대기 현금(w)과 그 원금(cbw) */
  var stateAt=function(b){
    var u=0,cb=0,w=0,cbw=0,dep=0,wd=0,pend=[],dead=false;
    var nextEntry=function(x){ for(var k=0;k<entries.length;k++) if(entries[k]>x) return entries[k]; return null; };
    var conv=function(x){ if(dead) return; pend=pend.filter(function(p){ if(p.inv!=null&&p.inv<=x&&(stopI==null||p.inv<=stopI)){ var c0=p.cost!=null?p.cost:p.amt; u+=p.amt/navF(p.inv); cb+=c0; w-=p.amt; cbw-=c0; return false; } return true; }); };
    /* 정리: 좌수를 그 봉의 기준가로 현금으로 바꾼다. 수동 정리(re)는 그 돈을 다음 진입의 대기금으로 두고 매입 원금도 이어 간다 */
    var doFlat=function(x,re){ if(u>1e-12){ var P=u*navF(x); w+=P; cbw+=cb; if(re) pend.push({amt:P,cost:cb,inv:nextEntry(x)}); u=0; cb=0; } if(!re) dead=true; };
    var flats=[]; mans.forEach(function(m){ flats.push({i:m,re:true}); }); if(windEnd!=null) flats.push({i:windEnd,re:false}); if(endBar!=null) flats.push({i:endBar,re:false});
    var bars=[]; ev.forEach(function(e){ if(e.i<=b&&bars.indexOf(e.i)<0) bars.push(e.i); });
    flats.forEach(function(f){ if(f.i<=b&&bars.indexOf(f.i)<0) bars.push(f.i); });
    bars.sort(function(a,b2){ return a-b2; });
    bars.forEach(function(x){
      ev.forEach(function(e){ if(e.i===x&&e.type==='add'){ dep+=e.amt; w+=e.amt; cbw+=e.amt; pend.push({amt:e.amt,inv:e.inv}); } });
      conv(x);
      flats.forEach(function(f){ if(f.i===x) doFlat(x,f.re); });
      ev.forEach(function(e){ if(e.i===x&&e.type!=='add'){
        var r=e.amt; wd+=e.amt;
        var fw=Math.min(r,Math.max(0,w)); /* 대기 현금부터 뺀다 */
        if(fw>0){ var k2=(w-fw)/w; cbw*=k2; pend.forEach(function(p){ p.amt*=k2; if(p.cost!=null) p.cost*=k2; }); w-=fw; r-=fw; }
        if(r>1e-9&&u>1e-12){ var nv=navF(x), sold=Math.min(u,r/nv); cb-=cb*(sold/u); u-=sold; } /* 나머지는 좌수를 판다. 평균 매입 원금도 같은 비율로 줄인다 */
      } });
    });
    conv(b);
    var nv2=navF(b), invested=u*nv2, value=invested+w;
    return {units:u,cb:cb,waiting:w,cbw:cbw,invested:invested,value:value,dep:dep,wd:wd,pnl:value+wd-dep,cost:cb+cbw};
  };
  var S=stateAt(T), total=S.pnl;
  var endI=endBar!=null?endBar:(windEnd!=null?windEnd:T), pnlPct=navAt(endI)/navAt(startI)-1; /* 같은 기간 전략 수익률: 넣고 뺀 돈과 무관 */
  var myPct=S.cost>1e-9?S.value/S.cost-1:0; /* 내 수익률: 평균 매입 기준가 대비 */
  var closed=(s2.r.trades||[]).filter(function(x){ var e=(x.exit!=null?x.exit:x.entry); return e>=startI&&e<=T; });
  var exits=closed.map(function(x){ return x.exit!=null?x.exit:x.entry; }).sort(function(a,b){ return a-b; });
  /* 실현 = 마지막 청산 봉까지의 손익. 미실현 = 그 뒤의 변화 (같은 척도라 가상 포지션이 생기지 않는다) */
  var realized=exits.length?stateAt(exits[exits.length-1]).pnl:0;
  var hwm=0; exits.forEach(function(x){ var p=stateAt(x).pnl; if(p>hwm) hwm=p; }); /* 분배는 실현 손익의 최고점을 넘긴 부분에만 (이중 부과 방지) */
  if(endBar!=null||windEnd!=null){ realized=total; if(total>hwm) hwm=total; }
  else if(mans.length){ mans.forEach(function(m){ var pm=stateAt(m).pnl; if(pm>hwm) hwm=pm; }); var lm=mans[mans.length-1]; if(!exits.length||exits[exits.length-1]<lm) realized=stateAt(lm).pnl; } /* 수동 정리도 손익을 확정한다 */
  var unreal=total-realized;
  var share=Math.max(0,hwm)*(m?m.share:0.1);
  var net=total-share;
  var hasOpen=s2.kind==='agent'?!!(st.open&&st.open.length):!!st.open;
  var posOpen=endBar==null&&windEnd==null&&S.units>1e-9&&hasOpen;
  var cashRatio=s2.kind==='agent'?(st.cash!=null?st.cash:1):(st.open?0:1);
  var est=S.value-share;
  var avail=Math.max(0,S.waiting+S.invested*(endBar!=null?1:cashRatio)-share);
  var lots=[]; ev.forEach(function(e){ lots[e.idx]={i:e.i,inv:e.inv}; });
  return {inv:inv,grossIn:grossIn,pnlPct:pnlPct,myPct:myPct,total:total,realized:realized,unreal:unreal,share:share,net:net,est:est,avail:avail,
    value:S.value,invested:S.invested,waiting:S.waiting,units:S.units,posOpen:posOpen,closedN:closed.length,immediate:immediate,
    entriesYear:entries.filter(function(i){ return i>T-365; }).length,lots:lots,at:stateAt,stopI:stopI,windEnd:windEnd,endBar:endBar};
}
