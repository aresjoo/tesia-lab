function cpMixNote(c2,d){
  var h='';
  if(c2.status!=='active') return h; /* 끝낸 전략 복사에는 기다리는 돈이 없다 */
  if(d.waiting>0.005) h+='<div class="cpx-sumline">투자 중 <b class="num">'+cpUsd(d.invested)+'</b>, 다음 진입 대기 <b class="num">'+cpUsd(d.waiting)+'</b>. 대기 중인 금액은 원본이 새로 진입할 때 함께 들어갑니다.</div>';
  var gap=(d.myPct-d.pnlPct)*100;
  if(d.grossIn>0&&Math.abs(gap)>=5) h+=cpAiHtml({w:0,t:'내 수익률('+mkPct0(d.myPct*100,1)+')과 같은 기간 전략 수익률('+mkPct0(d.pnlPct*100,1)+')이 다릅니다.   return h;
}

/* 청산 이력 탭 */
function cpxHist(c2,d){
  var s2=tfSSFind(c2.nick);
  var startI=c2.simStartI!=null?c2.simStartI:0, endI=c2.endI!=null?c2.endI:(c2.status==='closed'&&c2.flatI!=null?c2.flatI:1e15);
  var tr=(s2?s2.r.trades||[]:[]).filter(function(x){ var e=(x.exit!=null?x.exit:x.entry); return e>=startI&&e<=endI; }).slice(-30).reverse();
  if(!tr.length) return '<div class="cpp-empty"><b>아직 청산된 전략 복사 거래가 없습니다</b>첫 거래가 청산되면 진입가, 청산가, 손익이 여기에 쌓입니다.</div>';
  var wins=tr.filter(function(x){return x.pnl>0;}).length;
  var pu=function(x){ var a=d.at(x.entry); return x.pnl*a.invested*(x.w!=null?x.w:1); }; /* 그 거래에 들어간 내 금액 기준 */
  var sum=tr.reduce(function(a,x){return a+pu(x);},0);
  var KIND={sl:'손절 규칙',tp:'목표가 도달',time:'기간 청산'};
  return '<div class="cpx-sumline">청산 '+tr.length+'회, 그중 '+wins+'회 수익. 합계 <b class="num" style="color:'+(sum>=0?'#56c486':'#ee766a')+'">'+cpUsd(sum)+'</b>.</div>'
    +'<div class="cpx-tblw"><table class="cpx-tbl"><thead><tr>'
    +'<th>페어</th><th>방향</th><th>진입가</th><th>청산가</th><th>손익</th><th>수익률</th><th>청산 사유</th><th>청산 시각</th>'
    +'</tr></thead><tbody>'
    +tr.map(function(x){
      var pnlU=pu(x), up=x.pnl>=0;
      var PXB=mkPx(s2&&s2.p&&s2.p.px), eP=x.ep!=null?x.ep:(PXB[x.entry]||0), xP=x.xp!=null?x.xp:(PXB[x.exit]||eP);
      var dt=idxToDate(x.exit!=null?x.exit:x.entry);
      return '<tr><td>'+gEsc(c2.pairs[0])+'</td><td class="u">롱</td>'
        +'<td class="num">'+mkPxFmt(eP)+'</td><td class="num">'+mkPxFmt(xP)+'</td>'
        +'<td class="num '+(up?'u':'d')+'">'+cpUsd(pnlU)+'</td>'
        +'<td class="num '+(up?'u':'d')+'">'+(up?'+':'')+(x.pnl*100).toFixed(1)+'%</td>'
        +'<td>'+(KIND[x.kind]||'규칙 청산')+'</td>'
        +'<td class="num">'+(dt.getMonth()+1)+'. '+dt.getDate()+'.</td></tr>';
    }).join('')
    +'</tbody></table></div>';
}

/* 자금 이동 탭 — 원장 그대로 + 반복 입금 감지 (핵심 원칙 1) */
function cpxBal(c2,d){
  var rows=c2.ledger.map(function(e,ix){ return {e:e,ix:ix}; }).reverse();
  var adds=c2.ledger.filter(function(e){return e.type==='add';}).length;
  var warn=(adds>=3&&d.myPct<0)?cpAiHtml({w:1,t:'손실 구간에서 입금이 '+adds+'회 반복됐습니다. 반복 충전은 수익률을 실제보다 부풀려 보이게 합니다. 금액을 늘리기 전에 전략 자체를 다시 점검해보십시오.'}):'';
  /* 반영 시점: 그 입금이 좌수로 바뀐 봉. 아직이면 다음 진입 대기 (D34) */
  var when=function(r){ var l=(d.lots||[])[r.ix]; if(r.e.type!=='add'||!l) return '-'; if(l.inv==null||(d.stopI!=null&&l.inv>d.stopI)) return c2.status==='active'&&d.stopI==null?'다음 진입 대기':'투입 전 회수'; if(l.inv===l.i) return '바로'; return mkMD(l.inv)+' 진입 때'; };
  return warn
    +'<div class="cpx-sumline">예산 추가와 회수 '+rows.length+'건 (최근 180일).'+(c2.status==='active'&&d.waiting>0.005?' 다음 진입을 기다리는 금액 <b class="num">'+cpUsd(d.waiting)+'</b>.':'')+'</div>'
    +'<div class="cpx-tblw"><table class="cpx-tbl"><thead><tr>'
    +'<th>시각</th><th>유형</th><th>금액</th><th>자산</th><th>방향</th><th>반영</th>'
    +'</tr></thead><tbody>'
    +rows.map(function(r){
      var e=r.e, add=e.type==='add';
      return '<tr><td class="num">'+mkDT(e.at)+'</td>'
        +'<td>'+(add?'추가':'회수')+'</td><td class="num '+(add?'u':'')+'">'+(add?'+':'-')+cpUsd(e.amt)+'</td>'
        +'<td>USD</td><td>'+(add?'예산 → 전략 복사':'전략 복사 → 예산')+'</td><td>'+when(r)+'</td></tr>';
    }).join('')
    +'</tbody></table></div>';
}

function cpxPos(c2,d){

  if(!s2||!d.posOpen||c2.status!=='active')

  var c=s2.cfg||{}, st=(s2.r&&s2.r.state)||{}, o=st.open, list=!o?[]:o.length!=null?o:[o], up=d.unreal>=0;
  if(d.stopI!=null) list=list.filter(function(x){ return x.entry==null||x.entry<=d.stopI; }); /* 새 진입을 멈춘 뒤에 원본이 산 종목은 내 포지션이 아니다 */
