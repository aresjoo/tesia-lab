function Read-Header([string]$path) {
    $h = [ordered]@{ id = [IO.Path]::GetFileNameWithoutExtension($path); from = '?'; to = '?'; type = '?'; subject = ''; reply_to = '-'; created = '' }
    foreach ($line in (Get-Content -LiteralPath $path -Encoding UTF8 -TotalCount 20)) {
        if ($line -eq '---') { continue }
        if ($line -match '^([a-z_]+):\s*(.*)$') {
            $k = $Matches[1]
            if ($h.Contains($k)) { $h[$k] = $Matches[2].Trim() }
        } elseif ($line.Trim() -eq '') { break }
    }
    return [pscustomobject]$h
}

function Show-List($files, [string]$label) {
    if (-not $files -or $files.Count -eq 0) { Write-Host "$label : (?? ??)"; return }

    }
}

function New-Id([string]$from, [string]$subject) {
    # Keep Unicode letters so Korean subjects stay readable in the id.
    $slug = ($subject -replace '[^\p{L}\p{N}]+', '-').Trim('-').ToLower()
    if ($slug.Length -gt 40) { $slug = $slug.Substring(0, 40).Trim('-') }
    if (-not $slug) { $slug = 'msg' }
    return '{0}-{1}-{2}' -f (Get-Date -Format 'yyyyMMdd-HHmmss'), $from, $slug
}
